cmake_minimum_required(VERSION 3.16)
project(campus_marker_50032560 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs videoio highgui)
add_library(marker_detector detector.cpp)
target_include_directories(marker_detector PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(marker_detector PUBLIC ${OpenCV_LIBS})
target_compile_options(marker_detector PRIVATE -Wall -Wextra -Wpedantic)
add_executable(marker main.cpp)
target_link_libraries(marker PRIVATE marker_detector)
include(CTest)
if(BUILD_TESTING)
  add_executable(marker_tests tests.cpp)
  target_link_libraries(marker_tests PRIVATE marker_detector)
  add_test(NAME detector_regression COMMAND marker_tests)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../data/raw/marker_video.avi")
    add_test(NAME repository_video_regression COMMAND marker_tests "${CMAKE_CURRENT_SOURCE_DIR}/../../data/raw/marker_video.avi")
  endif()
endif()
